func (s *Service) Submit(ctx context.Context, in Report) (Incident, error) {if err := in.Validate(); err != nil {return Incident{}, err}tx, err := s.db.BeginTx(ctx, nil)if err != nil {return Incident{}, err}defer tx.Rollback()inc, err := s.store.Create(ctx, tx, in)if err != nil {return Incident{}, err}// same transaction: the event is durable, or nothing happenedif err := s.outbox.Add(ctx, tx, "incident.created", inc); err != nil {return Incident{}, err}if err := tx.Commit(); err != nil {return Incident{}, err}return inc, nil}
